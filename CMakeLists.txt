cmake_minimum_required(VERSION 3.25)
include(cmake/VcpkgTriplet.cmake)
project(bridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("#include <expected>
int main() { std::expected<int, int> value = std::unexpected(1); return value.error(); }"
  BRIDGE_HAS_STD_EXPECTED)
if(NOT BRIDGE_HAS_STD_EXPECTED)
  message(FATAL_ERROR "A C++23 compiler/library with std::expected is required. Use GCC 13+, or Clang 19+ when using GCC 13 libstdc++; configure clangd with the same compile_commands.json.")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(BRIDGE_BUILD_GUI "Build Qt Quick desktop" ON)
option(BRIDGE_BUILD_SESSION "Build Qt session libraries" ON)
option(BRIDGE_BUILD_FUZZER "Build Clang libFuzzer parser targets" OFF)
option(BRIDGE_WARNINGS_AS_ERRORS "Treat project warnings as errors" ON)
option(BRIDGE_CLANG_TIDY "Analyze project code during compilation" OFF)
option(BRIDGE_LINUX_SYSTEM_PACKAGE "Package against Ubuntu system Qt instead of collecting its runtime" OFF)
set(BRIDGE_SANITIZER "" CACHE STRING "address, undefined or thread")
include(CTest)
include(cmake/BridgeOptions.cmake)
function(bridge_library name)
  add_library(${name} STATIC ${ARGN})
  target_include_directories(${name} PUBLIC ${PROJECT_SOURCE_DIR}/include)
  bridge_options(${name})
endfunction()
bridge_library(bridge_core src/core/frame.cpp src/core/session.cpp src/core/diagnostics.cpp src/core/content.cpp src/core/file_protocol.cpp)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
bridge_library(bridge_security_crypto src/security/hash.cpp)
target_link_libraries(bridge_security_crypto PUBLIC bridge_core PRIVATE OpenSSL::Crypto)
bridge_library(bridge_io src/io/path.cpp)
if(UNIX)
  target_sources(bridge_io PRIVATE src/io/checkpoint_posix.cpp src/io/source_posix.cpp src/io/folder_posix.cpp)
  target_compile_definitions(bridge_io PRIVATE _FILE_OFFSET_BITS=64)
else()
  target_sources(bridge_io PRIVATE src/io/checkpoint_unsupported.cpp src/io/source_unsupported.cpp src/io/folder_unsupported.cpp)
endif()
target_link_libraries(bridge_io PUBLIC bridge_core PRIVATE bridge_security_crypto)
if(BRIDGE_BUILD_SESSION)
  find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)

  target_link_libraries(bridge_security PUBLIC bridge_core Qt6::Network PRIVATE bridge_security_crypto OpenSSL::Crypto)

  target_link_libraries(bridge_net PUBLIC bridge_core Qt6::Network PRIVATE bridge_security)

  bridge_library(bridge_app src/app/session.cpp include/bridge/app/session.hpp

  target_link_libraries(bridge_app PUBLIC bridge_core bridge_net bridge_io Qt6::Core PRIVATE Threads::Threads)

if(BUILD_TESTING)

  set_tests_properties(unit PROPERTIES LABELS unit TIMEOUT 30)
  add_executable(bridge_io_tests tests/integration/checkpoint.cpp)

  if(BRIDGE_BUILD_SESSION)
